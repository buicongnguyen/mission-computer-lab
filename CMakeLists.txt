cmake_minimum_required(VERSION 3.16)
project(mission_computer_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(mission_core src/supervisor.cpp)
target_include_directories(mission_core PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(mission_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(ENABLE_SANITIZERS)
    target_compile_options(mission_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mission_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(mission_supervisor src/main.cpp)
target_link_libraries(mission_supervisor PRIVATE mission_core)
enable_testing()
add_executable(test_supervisor tests/test_supervisor.cpp)
target_link_libraries(test_supervisor PRIVATE mission_core)
add_test(NAME supervisor_contract COMMAND test_supervisor)
